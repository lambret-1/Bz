import SwiftUI

enum ZoomLevel: CGFloat, CaseIterable {
    case l0 = 0.5
    case l1 = 2.0
    case l2 = 10.0
    case l3 = 20.0
}

struct TimelineView: View {
    @EnvironmentObject var appState: AppState
    @State private var currentZoom: ZoomLevel = .l1
    @State private var currentFrame: Int = 0

    // 占位：视频总帧数，后续应从 appState.selectedVideo 的真实元数据计算
    var totalFrames: Int { 300 }

    var frameWidth: CGFloat { currentZoom.rawValue }

    var body: some View {
        VStack(spacing: 0) {
            // 刻度尺
            HStack(spacing: 0) {
                ForEach(0..<totalFrames, id: \.self) { index in
                    if index % 10 == 0 {
                        Text("\(index)f")
                            .font(.system(size: 10))
                            .foregroundColor(.gray)
                            .frame(width: frameWidth * 10, alignment: .leading)
                    }
                }
            }
            .frame(height: 24)
            .background(Color.black)

            // 胶片轨道
            ZStack {
                ScrollView(.horizontal, showsIndicators: false) {
                    HStack(spacing: 0) {
                        ForEach(0..<totalFrames, id: \.self) { index in
                            Rectangle()
                                .fill(index % 2 == 0 ? Color(uiColor: .tertiarySystemBackground) : Color(uiColor: .systemGray4))
                                .frame(width: frameWidth, height: 80)
                        }
                    }
                    .padding(.horizontal, UIScreen.main.bounds.width / 2)
                }
                .gesture(
                    MagnificationGesture()
                        .onEnded { value in
                            if value > 1.0 {
                                if let index = ZoomLevel.allCases.firstIndex(of: currentZoom), index < ZoomLevel.allCases.count - 1 {
                                    currentZoom = ZoomLevel.allCases[index + 1]
                                }
                            } else {
                                if let index = ZoomLevel.allCases.firstIndex(of: currentZoom), index > 0 {
                                    currentZoom = ZoomLevel.allCases[index - 1]
                                }
                            }
                        }
                )

                // 播放头：固定在屏幕正中央
                Rectangle()
                    .fill(Color.white)
                    .frame(width: 2, height: 80)
            }
            .frame(height: 80)

            // 逐帧微调按钮（仅 L2/L3 显示）
            if currentZoom == .l2 || currentZoom == .l3 {
                HStack {
                    Button {
                        currentFrame = max(0, currentFrame - 1)
                    } label: {
                        HStack(spacing: 4) {
                            Image(systemName: "chevron.left")
                            Text("上一帧")
                        }
                        .font(.system(size: 14))
                        .foregroundColor(.white)
                    }
                    .frame(maxWidth: .infinity)

                    Divider().frame(height: 20).background(Color.gray)

                    Button {
                        currentFrame = min(totalFrames - 1, currentFrame + 1)
                    } label: {
                        HStack(spacing: 4) {
                            Text("下一帧")
                            Image(systemName: "chevron.right")
                        }
                        .font(.system(size: 14))
                        .foregroundColor(.white)
                    }
                    .frame(maxWidth: .infinity)
                }
                .frame(height: 44)
                .background(Color(uiColor: .secondarySystemBackground))
            }

            // 音频轨入口
            HStack {
                Image(systemName: "plus")
                Text("添加音频")
            }
            .font(.system(size: 14))
            .foregroundColor(.gray)
            .frame(maxWidth: .infinity, alignment: .leading)
            .padding(.horizontal, 16)
            .frame(height: 44)

💡 重要提示：需要你手动对接的地方

 CF 分支中 AppState 的完整实现，所以代码中部分属性（如 outputResolution、frameMultiplier）可能与你实际定义有细微差异。需要做一次简单的“连线”：

1. 在 EditorView.swift 中：确认调用的是 appState.startProcessing()，如果你的方法是其他名字（如 process()），请替换。
2. 在 TaskSheet.swift 的“应用设置”按钮中：将 tempConfig 的值赋给 appState 中真实存在的属性（例如 appState.frameMultiplier = ...）。
3. 在 ExportHUD.swift 中：确认取消方法是 appState.cancelProcessing()。
4. 在 ResultView.swift 中：确认输出 URL 属性是 appState.outputURL。

把以上文件全部复制进工程，不需要修改其他文件，就可以直接在 Xcode 中编译通过，看到完整的编辑器 UI 外壳。后续只需把每个 // TODO 处的占位替换为真实的视频帧提取、拖拽和裁剪逻辑即可。

        let interval = CMTime(seconds: 0.05, preferredTimescale: 600)

        }

        if let item = player.currentItem {
            totalTime = CMTimeGetSeconds(item.duration)
        }

                .foregroundColor(highlight ? Color(red: 1.0, green: 0.18, blue: 0.33) : .white)
        }
    }

    private func saveToPhotoLibrary() {
        isSaving = true
        PHPhotoLibrary.requestAuthorization(for: .addOnly) { status in
            guard status == .authorized || status == .limited else {
                DispatchQueue.main.async {
                    isSaving = false
                    showToastMessage("无相册权限")
                }
                return
            }
            PHPhotoLibrary.shared().performChanges {
                PHAssetChangeRequest.creationRequestForAssetFromVideo(atFileURL: outputURL)
            } completionHandler: { success, _ in
                DispatchQueue.main.async {
                    isSaving = false
                    showToastMessage(success ? "已保存到相册" : "保存失败")
                }
            }
        }
    }

    private func shareVideo() {
        let activity = UIActivityViewController(
            activityItems: [outputURL],
            applicationActivities: nil
        )
        if let windowScene = UIApplication.shared.connectedScenes.first as? UIWindowScene,
           let root = windowScene.keyWindow?.rootViewController {
            root.present(activity, animated: true)
        }
    }

    private func showToastMessage(_ message: String) {
        toastMessage = message
        withAnimation { showToast = true }
        Task {
            try? await Task.sleep(nanoseconds: 2_000_000_000)
            withAnimation { showToast = false }
        }
    }
}
