                    // 全屏逻辑
                } label: {
                    Image(systemName: "arrow.up.left.and.arrow.down.right")
                        .font(.system(size: 20))
                        .foregroundColor(.topBarText)
                }
            }
        }
        .padding(.horizontal, 16)
        .frame(height: 48)
        .background(Color.topBarBg)
        .onAppear { setupPlayerObservers() }
        .onDisappear { removePlayerObservers() }
    }

    private func togglePlayback() {
        guard let player = PlayerManager.shared.player else { return }
        if isPlaying {
            player.pause()
        } else {
            player.play()
        }
        isPlaying.toggle()
    }

    private func setupPlayerObservers() {
        guard let player = PlayerManager.shared.player else { return }
        let interval = CMTime(seconds: 0.05, preferredTimescale: 600)
        timeObserver = player.addPeriodicTimeObserver(forInterval: interval, queue: .main) { time in
            currentTime = CMTimeGetSeconds(time)
        }
        if let item = player.currentItem {
            totalTime = CMTimeGetSeconds(item.duration)
        }
    }

    private func removePlayerObservers() {
        if let observer = timeObserver, let player = PlayerManager.shared.player {
            player.removeTimeObserver(observer)
        }
        timeObserver = nil
    }

    private func undo() {
        guard let last = undoStack.popLast() else { return }
        redoStack.append(last)
        // TODO: 应用到 AppState.currentProject.segments
    }

    private func redo() {
        guard let last = redoStack.popLast() else { return }
        undoStack.append(last)
        // TODO: 应用到 AppState.currentProject.segments
    }
}

import SwiftUI

struct TimecodeDisplay: View {
    let currentFrame: Int
    let totalFrames: Int
    let currentTime: TimeInterval
    let totalTime: TimeInterval
    let isFrameMode: Bool
    let thumbnail: UIImage?

    var body: some View {
        HStack(spacing: 8) {
            if isFrameMode {
                Text("f\(String(format: "%03d", currentFrame)) / f\(String(format: "%03d", totalFrames))")
                    .font(.system(size: 14, weight: .bold, design: .monospaced))
                    .foregroundColor(Color.accentPink)

                if let thumbnail {
                    Image(uiImage: thumbnail)
                        .resizable()
                        .scaledToFill()
                        .frame(width: 40, height: 40)
                        .cornerRadius(4)
                        .clipped()
                }
            } else {
                Text(formatTime(currentTime) + " / " + formatTime(totalTime))
                    .font(.system(size: 14, design: .monospaced))
                    .foregroundColor(.topBarText)
            }
        }
    }

    private func formatTime(_ time: TimeInterval) -> String {
        let minutes = Int(time) / 60
        let seconds = Int(time) % 60
        let ms = Int((time - floor(time)) * 1000)
        return String(format: "%02d:%02d.%03d", minutes, seconds, ms)
    }
}

enum ZoomLevel: CGFloat, CaseIterable {

import SwiftUI
import AVFoundation

struct AudioWaveformView: View {
    let url: URL
    let duration: TimeInterval
    let color: Color

    @State private var samples: [Float] = []

    var body: some View {
        GeometryReader { geometry in
            HStack(spacing: 1) {
                ForEach(Array(samples.enumerated()), id: \.offset) { _, sample in
                    let barHeight = max(2, CGFloat(sample) * geometry.size.height)
                    RoundedRectangle(cornerRadius: 1)
                        .fill(color.opacity(0.7))
                        .frame(width: 2, height: barHeight)
                }
            }
            .frame(maxHeight: .infinity, alignment: .center)
        }
        .onAppear { loadWaveform() }
    }

    private func loadWaveform() {
        guard let file = try? AVAudioFile(forReading: url) else { return }
        let format = file.processingFormat
        let frameCount = UInt32(file.length)
        guard let buffer = AVAudioPCMBuffer(pcmFormat: format, frameCapacity: frameCount) else { return }
        try? file.read(into: buffer)
        guard let channelData = buffer.floatChannelData else { return }

        let samplesPerPixel = max(1, Int(frameCount) / 200)
        var result: [Float] = []

        for i in stride(from: 0, to: Int(frameCount), by: samplesPerPixel) {
            var maxValue: Float = 0
            for j in 0..<samplesPerPixel {
                if i + j < Int(frameCount) {
                    maxValue = max(maxValue, abs(channelData[0][i + j]))
                }
            }
            result.append(maxValue)
        }
        samples = result
    }
}
