本文档描述如何将 PiperSR 2x 超分模型集成到 iOS 视频增强工作台中，实现独立超分模式。核心内容包括：

· 最低支持 iOS 16.0
· 完全端侧运行，不联网
· 无设备分级，所有参数由用户手动选择
· 补帧与超分互斥，不可同时使用

二、模型简介与核心参数

2.1 模型简介

PiperSR 是首个专为 Apple Neural Engine（ANE）从零设计的开源超分辨率模型，由 ModelPiper 团队开发。与为 GPU 设计的 Real-ESRGAN、ESPCN 不同，PiperSR 的每一层操作都经过验证可在 ANE 上原生执行，不存在静默回退到 GPU/CPU 的情况。

2.2 核心参数

参数 值
参数量 453K
CoreML FP16 包体 928 KB
放大倍数 2x
PSNR (Set5) 37.54 dB
架构 6 个残差块，64 通道，SiLU 激活，PixelShuffle 上采样
输入尺寸 640×360
输出尺寸 1280×720
ANE 推理延迟 约 22.5 ms（M2 Max 实测）
视频吞吐量 44.4 FPS（M2 Max，360p→720p）
许可证 代码 AGPL-3.0，权重 PiperSR Model License

4.2 模式枚举 PipelineMode.swift

```swift
import Foundation

enum PipelineMode: Equatable {
    case interpolation(FrameMultiplier)
    case superResolution(SuperResolutionEngineType)

    var displayName: String {
        switch self {
        case .interpolation(let multiplier):
            return "补帧 \(multiplier.rawValue)x"
        case .superResolution(let engine):
            return "超分 \(engine.displayName)"
        }
    }
}

enum FrameMultiplier: Int, CaseIterable, Identifiable {
    case x2 = 2
    case x4 = 4
    var id: Int { rawValue }
    var label: String { "\(rawValue)x" }
}

enum SuperResolutionEngineType: String, CaseIterable, Identifiable {
    case piperSR = "PiperSR"
    case realESRGAN = "Real-ESRGAN"

    var id: String { rawValue }

    var displayName: String {
        switch self {
        case .piperSR: return "PiperSR 2x"
        case .realESRGAN: return "Real-ESRGAN 4x"
        }
    }

    var scale: Int {
        switch self {
        case .piperSR: return 2
        case .realESRGAN: return 4
        }
    }

    var maxInputWidth: Int {
        switch self {
        case .piperSR: return 1280
        case .realESRGAN: return 854
        }
    }
}

enum SuperResolutionError: LocalizedError {
    case modelNotLoaded
    case inputSizeMismatch
    case inferenceFailed
    case bufferPoolExhausted
    case outputClampFailed

    var errorDescription: String? {
        switch self {
        case .modelNotLoaded: return "超分模型未加载"
        case .inputSizeMismatch: return "输入尺寸与模型不匹配"
        case .inferenceFailed: return "超分推理失败"
        case .bufferPoolExhausted: return "缓冲区耗尽"
        case . outputClampFailed: return "输出钳制失败"
        }
    }
}
```

4.4 PiperSR 推理引擎 PiperSREngine.swift

```swift
import CoreML
import CoreVideo
import Metal
import Accelerate

final class PiperSREngine: SuperResolutionEngine {

    static let inputWidth  = 640
    static let inputHeight = 360
    static let outputWidth  = 1280
    static let outputHeight = 720
    let scale: Int = 2
    var inputSize: CGSize { CGSize(width: Self.inputWidth, height: Self.inputHeight) }

    private let model: MLModel
    private let device: MTLDevice
    private let commandQueue: MTLCommandQueue
    private let preprocessPipeline: MTLComputePipelineState
    private let postprocessPipeline: MTLComputePipelineState
    private let pool: PixelBufferPool

    init?() {
        guard let modelURL = Bundle.main.url(
            forResource: "PiperSR_2x_video_720p",
            withExtension: "mlmodelc"
        ) else {
            print("[PiperSR] ⚠️ 未找到模型文件")
            return nil
        }

        let config = MLModelConfiguration()
        config.computeUnits = .all

        guard let model = try? MLModel(contentsOf: modelURL, configuration: config) else {
            print("[PiperSR] ⚠️ 模型加载失败")
            return nil
        }
        self.model = model

        guard let device = MTLCreateSystemDefaultDevice(),
              let queue = device.makeCommandQueue(),
              let library = device.makeDefaultLibrary(),
              let preFn = library.makeFunction(name: "bgra_to_nchw"),
              let postFn = library.makeFunction(name: "nchw_to_bgra") else {
            print("[PiperSR] ⚠️ Metal 初始化失败")
            return nil
        }

        guard let pre = try? device.makeComputePipelineState(function: preFn),
              let post = try? device.makeComputePipelineState(function: postFn) else {
            print("[PiperSR] ⚠️ Metal 管线创建失败")
            return nil
        }

        self.device = device
        self.commandQueue = queue
        self.preprocessPipeline = pre
        self.postprocessPipeline = post

        self.pool = PixelBufferPool()
        pool.configure(width: Self.outputWidth, height: Self.outputHeight)
    }

    func upscale(_ pixelBuffer: CVPixelBuffer) -> Result<CVPixelBuffer, SuperResolutionError> {
        let w = CVPixelBufferGetWidth(pixelBuffer)
        let h = CVPixelBufferGetHeight(pixelBuffer)
        guard w == Self.inputWidth, h == Self.inputHeight else {
            return .failure(.inputSizeMismatch)
        }

        guard let inputArray = makeInputArray(from: pixelBuffer) else {
            return .failure(.inferenceFailed)
        }

        guard let prediction = try? model.prediction(from: ["input": inputArray]),
              let outputArray = prediction.featureValue(for: "output")?.multiArrayValue else {
            return .failure(.inferenceFailed)
        }

        guard let outputBuffer = pool.makeBuffer() else {
            return .failure(.bufferPoolExhausted)
        }
        guard convertToPixelBuffer(outputArray, into: outputBuffer) else {
            return .failure(.inferenceFailed)
        }

        return .success(outputBuffer)
    }

    private func makeInputArray(from pixelBuffer: CVPixelBuffer) -> MLMultiArray? {
        guard let array = try? MLMultiArray(
            shape: [1, 3, NSNumber(value: Self.inputHeight), NSNumber(value: Self.inputWidth)],
            dataType: .float16
        ) else { return nil }

        CVPixelBufferLockBaseAddress(pixelBuffer, .readOnly)
        defer { CVPixelBufferUnlockBaseAddress(pixelBuffer, .readOnly) }

        guard let base = CVPixelBufferGetBaseAddress(pixelBuffer) else { return nil }
        let bytesPerRow = CVPixelBufferGetBytesPerRow(pixelBuffer)
        let plane = Self.inputWidth * Self.inputHeight
        let ptr = array.dataPointer.bindMemory(to: Float16.self, capacity: plane * 3)

        for y in 0..<Self.inputHeight {
            for x in 0..<Self.inputWidth {
                let offset = y * bytesPerRow + x * 4
                let b = Float16(base.load(fromByteOffset: offset,     as: UInt8.self)) / 255.0
                let g = Float16(base.load(fromByteOffset: offset + 1, as: UInt8.self)) / 255.0
                let r = Float16(base.load(fromByteOffset: offset + 2, as: UInt8.self)) / 255.0
                let idx = y * Self.inputWidth + x
                ptr[idx]                 = r
                ptr[plane + idx]         = g
                ptr[2 * plane + idx]     = b
            }
        }
        return array
    }

    private func convertToPixelBuffer(_ array: MLMultiArray, into buffer: CVPixelBuffer) -> Bool {
        let plane = Self.outputWidth * Self.outputHeight
        let ptr = array.dataPointer.bindMemory(to: Float16.self, capacity: plane * 3)

        CVPixelBufferLockBaseAddress(buffer, [])
        defer { CVPixelBufferUnlockBaseAddress(buffer, []) }

        guard let base = CVPixelBufferGetBaseAddress(buffer) else { return false }
        let bytesPerRow = CVPixelBufferGetBytesPerRow(buffer)

        for y in 0..<Self.outputHeight {
            for x in 0..<Self.outputWidth {
                let idx = y * Self.outputWidth + x
                let r = Float(ptr[idx])
                let g = Float(ptr[plane + idx])
                let b = Float(ptr[2 * plane + idx])

                let offset = y * bytesPerRow + x * 4
                base.storeBytes(of: UInt8(clamping: Int(b * 255)), toByteOffset: offset,     as: UInt8.self)
                base.storeBytes(of: UInt8(clamping: Int(g * 255)), toByteOffset: offset + 1, as: UInt8.self)
                base.storeBytes(of: UInt8(clamping: Int(r * 255)), toByteOffset: offset + 2, as: UInt8.self)
                base.storeBytes(of: UInt8(255),                     toByteOffset: offset + 3, as: UInt8.self)
            }
        }
        return true
    }
}
```

五、管线集成

5.1 VideoPipeline 修改

在 VideoPipeline 中增加 mode 参数，根据模式分发到补帧或超分。

```swift
func process(
    inputURL: URL,
    outputURL: URL,
    mode: PipelineMode,
    progress: @escaping (Double, ProcessingStage) -> Void,
    completion: @escaping (Result<URL, Error>) -> Void
) {
    switch mode {
    case .interpolation(let multiplier):
        runInterpolation(inputURL: inputURL, outputURL: outputURL, multiplier: multiplier, progress: progress, completion: completion)
    case .superResolution(let engineType):
        runSuperResolution(inputURL: inputURL, outputURL: outputURL, engineType: engineType, progress: progress, completion: completion)
    }
}
```

5.2 超分模式执行顺序

```
解码 → 尺寸适配（缩放至 640×360）→ PiperSR 超分 → 缩放至目标分辨率 → 编码（保持原帧率）
```

注意：

· 帧率不变。30fps 输入，输出仍为 30fps。
· PTS 不变。沿用原始时间戳，不重新计算。
· 音频透传。使用与补帧模式相同的音频处理逻辑。

5.3 分辨率适配策略

PiperSR 的固定输入尺寸为 640×360。如果用户的视频分辨率不是这个尺寸，需要先缩放到 640×360，超分到 1280×720，再缩放到用户期望的分辨率（如 1920×1080）。

推荐策略：在 UI 上固定超分模式为 “720p 增强”，即输入 720p 视频，输出 1080p 视频。

· 若输入 > 720p：提示“视频已经是高清，无需超分”。
· 若输入 < 720p：先缩放到 640×360，超分到 1280×720，再缩放到。

5.4 与补帧互斥

在 UI 层（TaskSheet）中，使用单选组保证用户只能选择补帧或超分之一。VideoPipeline 内部也需进行校验，防止误传参数。

十、注意事项

1. 不要同时串联 PiperSR 与 RIFE。用户在 UI 上选择超分后，应跳过补帧逻辑。
2. 输入尺寸固定为 640×360。若原视频不是该尺寸，需先缩放，这会增加一次图像处理开销。
3. 音频处理逻辑完全复用现有方案，不需要为超分模式单独开发音频管线。
4. 码率动态调整。超分后分辨率提升，数据量增加，输出码率应相应提高（如 1.5 倍），否则画质提升会被压缩抵消。
